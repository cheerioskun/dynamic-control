cmake_minimum_required(VERSION 3.20)
project(dynamic_control LANGUAGES C CXX)

include(CTest)
option(DC_BUILD_BENCH "Build the three-path Google Benchmark fixture" OFF)
add_subdirectory(runtime)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(dc_runtime_test tests/runtime.cc)
  target_link_libraries(dc_runtime_test PRIVATE dynamic_control Threads::Threads)
  add_test(NAME runtime COMMAND dc_runtime_test)
  add_executable(dc_msem_mock_test tests/msem_mock.cc)
  target_compile_features(dc_msem_mock_test PRIVATE cxx_std_17)
  target_include_directories(dc_msem_mock_test PRIVATE runtime)
  target_link_libraries(dc_msem_mock_test PRIVATE Threads::Threads)
  add_test(NAME msem_mock COMMAND dc_msem_mock_test)
endif()
if(DC_BUILD_BENCH)
  add_subdirectory(bench)
endif()
